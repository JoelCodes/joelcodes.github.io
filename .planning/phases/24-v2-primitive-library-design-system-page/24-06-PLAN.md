    ```typescript
        // -------------------------------------------------------------------------
        // Hover lift magnitude — closes UAT Gap 2 (plan 24-06). The interactive
        // Card must produce a non-zero translateY on :hover (was previously -2px,
        // below perception threshold — raised to -4px / -translate-y-1 per UAT
        // fix). Regression guard: any future change that re-introduces an
        // imperceptible or zero lift will fail this assertion.
        // See .planning/debug/interactive-card-no-hover-lift.md for diagnosis.
        //
        // Robustness notes:
        //   (a) Test 7 above called `await interactiveCard.focus()` — focus state
        //       can produce sub-pixel layout drift on Chromium and (more
        //       importantly) means document.activeElement === the card, which
        //       confuses subsequent mouse-baseline reads. Explicitly blur the
        //       active element before establishing the no-hover baseline.
        //   (b) Computed `transform` is the most stable cross-browser readout
        //       for Tailwind's `translate` utility — Chromium emits the value
        //       as `matrix(1, 0, 0, 1, tx, ty)` where index 5 is the Y
        //       translation in px. Parsing the matrix tuple avoids the regex
        //       fragility of the `translate` property (which Chromium may emit
        //       as `"X Y"`, `"Y"` alone, or `"none"` depending on the rule).
        // -------------------------------------------------------------------------
        // (a) Clear residual focus so it doesn't bias baseline measurement.
        await page.evaluate(() => {
          const active = document.activeElement as HTMLElement | null;
          if (active && typeof active.blur === 'function') active.blur();
        });
        await page.mouse.move(0, 0);

        // Baseline (no hover, no focus) — translateY component of the
        // computed `transform` matrix should be 0.
        const parseMatrixTy = (transform: string): number => {
          if (!transform || transform === 'none') return 0;
          // matrix(a, b, c, d, tx, ty)  — index 5 is ty
          // matrix3d(...) — index 13 is ty, but Tailwind `translate` produces
          // 2D matrix, so handle the 6-arg form only.
          const inside = transform.replace(/^matrix(?:3d)?\(/, '').replace(/\)$/, '');
          const parts = inside.split(',').map((s) => parseFloat(s.trim()));
          if (parts.length === 6) return parts[5];        // 2D matrix → ty
          if (parts.length === 16) return parts[13];      // matrix3d → ty
          return Number.NaN;
        };

        const baselineTransform = await interactiveCard.evaluate(
          (el) => window.getComputedStyle(el as Element).transform
        );
        const baselineTy = parseMatrixTy(baselineTransform);
        // Baseline ty must be 0 (or close to it — allow ±0.5px sub-pixel slack)
        expect(Math.abs(baselineTy)).toBeLessThanOrEqual(0.5);

        // Hover the card and re-read the computed transform matrix.
        await interactiveCard.hover();
        const hoverTransform = await interactiveCard.evaluate(
          (el) => window.getComputedStyle(el as Element).transform
        );
        const hoverTy = parseMatrixTy(hoverTransform);

        // Assertion: ty must be meaningfully negative (≤ -3px guards both -4
        // and -6; the broken -2px state would fail this assertion).
        expect(hoverTy).toBeLessThanOrEqual(-3);
    ```

    - DO NOT use the `translate` CSS property (`getComputedStyle(el).translate`) — Chromium reports it inconsistently (`"X Y"`, `"Y"` alone, `"none"`) and the regex approach in earlier drafts was fragile. The matrix tuple is canonical.
